07Modules

Change the app from inside the app.

Every feature in Loom is a module. Yours is a folder of .cs files: save it, and Loom compiles it and swaps it in while it runs. Install modules other people wrote from the community hub, or send yours.

NotesModule.cs
public sealed class NotesModule : ILoknModule
{
	public string Id => "notes";
	public string Title => "Notes";
	public void Register( ModuleBuilder builder )
	{
		builder.Panel( "note", "Note", context => new NoteContent( context ) );
	}
}
internal sealed class NoteContent( PanelContext context ) : PanelContent
{
	private readonly State<string> _text = new( "" );
	public override Blob Build() => new Column
	{
		Children =
		[
			new TextEntry { Value = _text.Value, Placeholder = "Write a note…" },
			// …
		],
	};
}
